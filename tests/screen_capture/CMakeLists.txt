cmake_minimum_required(VERSION 3.20)
project(screen_capture_regression LANGUAGES CXX)
enable_testing()
add_executable(test_screen_capture test_screen_capture.cpp)
target_compile_features(test_screen_capture PRIVATE cxx_std_20)
target_compile_definitions(test_screen_capture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(test_screen_capture PRIVATE /W4 /WX)
endif()
target_include_directories(test_screen_capture PRIVATE ../../src/Core)
add_test(NAME screen_capture_regression COMMAND test_screen_capture)
